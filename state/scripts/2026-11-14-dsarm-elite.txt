import csv
# Example: Pulling data from a CSV file
with open('security_logs.csv', 'r') as file:
    reader = csv.DictReader(file)
    for row in reader:
        ip = row.get('source_ip', 'unknown')
        if ip.startswith('192.168.'):
            print(f'Flagging internal IP: {ip}')

# Example: Parsing a simple log file line
log_line = '2023-10-27 10:05:13 ALERT SSH brute-force attempt from 1.2.3.4'
if 'brute-force' in log_line:
